\section*{Exercise 2.6(E.2.6, page 115)}
Show that the following orthogonality relation
\begin{equation*}
    \int \left[ \chi_{m,k} \left( \frac{\hbar(k+k')}{2} + eBy \right) \chi_{n,k'} \right] \dd y = \delta_{k,k'}
\end{equation*}
where the two function $\chi_{m,k}(y)$ and $\chi_{n,k'}(y)$ satisfy
\begin{equation*}
    \begin{aligned}
        & \left[ E_s + \frac{(\hbar k + eBy)^2}{2 m} + \frac{p_y^2}{2m} + U(y) \right] \chi_{m,k}(y) = E\chi_{m,k}(y) \\
        & \left[ E_s + \frac{(\hbar k' + eBy)^2}{2 m} + \frac{p_y^2}{2m} + U(y) \right] \chi_{n,k'}(y) = E\chi_{n,k'}(y)
    \end{aligned}
\end{equation*}
\begin{proof}
    Subsituting $\chi_{n,k'}(y)$ and $\chi_{m,k}(y)$ into the Schrödinger equation, we have
    \begin{equation*}
        \begin{aligned}
            & \chi_{n,k'}(y) \left[ E_s + \frac{(\hbar k + eBy)^2}{2 m} + \frac{p_y^2}{2m} + U(y) \right] \chi_{m,k}(y) = E\chi_{m,k}(y) \\
            & \chi_{m,k}(y) \left[ E_s + \frac{(\hbar k' + eBy)^2}{2 m} + \frac{p_y^2}{2m} + U(y) \right] \chi_{n,k'}(y) = E\chi_{n,k'}(y)
        \end{aligned}
    \end{equation*}
    Subtracting the two equations, we get
    \begin{equation*}
        \begin{aligned}
            & \chi_{n,k'}(y) \left[ E_s + \frac{(\hbar k + eBy)^2}{2 m} + \frac{p_y^2}{2m} + U(y) \right] \chi_{m,k}(y) \\
            - &\chi_{m,k}(y) \left[ E_s + \frac{(\hbar k' + eBy)^2}{2 m} + \frac{p_y^2}{2m} + U(y) \right] \chi_{n,k'}(y) \\
            = & E\chi_{m,k}(y)\chi_{n,k'}(y) - E\chi_{n,k'}(y)\chi_{m,k}(y) \\
            \Rightarrow & \chi_{n,k'}(y) \frac{(\hbar k + eBy)^2-(\hbar k' + eBy)^2}{2 m} \chi_{m,k}(y) = 0 \\
            \Rightarrow & \chi_{n,k'}(y) \frac{\hbar(k-k')(\hbar(k+k') + 2eBy)}{2 m} \chi_{m,k}(y) = 0
        \end{aligned}
    \end{equation*}
    Integrating both sides over $y$, we have
    \begin{equation*}
        \int \chi_{n,k'}(y) \frac{\hbar(k+k') + 2eBy}{2 m} \chi_{m,k}(y) \dd y = 0, \quad \text{if } k \neq k'
    \end{equation*}
    By nomailzing the wave function in appropriate way, we have
    \begin{equation*}
        \int \chi_{m,k} \left( \frac{\hbar(k+k')}{2} + eBy \right) \chi_{n,k'} \dd y = \delta_{k,k'}
    \end{equation*}
\end{proof}
