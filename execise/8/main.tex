\section*{Exercise 3.6}
Derive
\begin{equation*}
    \hat{G}^R(\tau) = -\frac{\ii}{\hbar}\theta(\tau)\eexp^{-\frac{\ii}{\hbar}E_n\tau}\ket{\psi_n}\bra{\psi_n}
\end{equation*}
using the following expansion
\begin{equation*}
    \hat{G}^R(E) = \frac{1}{(E + \ii \eta)\hat{I}-\hat{H}}\sum_n \ket{\psi_n}\bra{\psi_n} = \frac{1}{E - E_n + \ii\eta}\sum_n \ket{\psi_n}\bra{\psi_n}
\end{equation*}
\section*{Execise 3.7}
For 1D free electron
\begin{equation*}
    \left[ E \pm \ii \eta + \frac{\hbar^2}{2m}\pdv{^2}{x^2} \right]G^{R/A}(x,x';E) = \delta(x-x')
\end{equation*}
Find two exact solution: $G_1(x-x',E) = G^R$ corresponding to "+", and $G_2(x-x',E) = G^A$ corresponding to "-".
