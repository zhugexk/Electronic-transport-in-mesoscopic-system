\section*{Exercise 3.8(E.3.2, page 171)}
Derive Eq.3.3.15 
\begin{equation*}
    G^R_E(x,y;x',y') = \sum_m -\frac{\ii}{\hbar v_m} \chi_m(y)\chi_m(y') \eexp^{\ii k_m \abs{x-x'}}
\end{equation*}
where $k_m = \frac{\sqrt{2m(E-\epsilon_{m,0})}}{\hbar}$ and $v_m = \frac{\hbar k}{m}$. For the Green's function of an infinite wire using a eigenfunctioin expansion Eq.3.3.17.
\begin{equation*}
    G^R(\vec{r},\vec{r}') = \sum_\alpha \frac{\psi_\alpha(\vec{r})\psi_\alpha^*(\vec{r}')}{E - \epsilon_\alpha + \ii \eta}
\end{equation*}

\section*{Execise 3.9(E.3.3, page 171)}
Show that the Green's function for a semi-infinite lead $p$ can be written as
\begin{equation*}
    g^R_p(p_i,p_j) = -\frac{1}{t} \sum_m \chi_m(p_i)\exp(+\ii k_m a)\chi_m (p_j)
\end{equation*}
where the index $m$ represents the propagating modes in lead $p$.
